#include <tunables/global>

profile resolved_watchdog flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # full_access + host_dbus: talks to the host system bus to restart
  # systemd-resolved when the stub resolver hangs.
  capability,
  file,
  signal,
  network tcp,
  network udp,
  network inet,
  network inet6,
  network unix dgram,
  network unix stream,
  network netlink raw,

  capability setgid,
  capability setuid,
  capability dac_read_search,
  capability sys_admin,

  /ha-entrypoint.sh rix,
  /bin/** rix,
  /usr/bin/** rix,
  /usr/local/bin/** rix,
  /usr/bin/nslookup rix,
  /usr/bin/timeout rix,
  /usr/bin/dbus-send rix,
  /usr/bin/jq rix,

  /data/** rw,
  /tmp/** mrwk,
  /var/run/{,**} mrwkl,
  /run/{,**} rwk,
  /dev/tty rw,

  # Host D-Bus (system bus) for RestartUnit on systemd-resolved
  /run/dbus/system_bus_socket rw,
  /var/run/dbus/system_bus_socket rw,
  /etc/machine-id r,

  /etc/nsswitch.conf r,
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/localtime r,
  /etc/passwd r,
  /etc/group r,
  /usr/share/zoneinfo/** r,
  /proc/sys/net/** r,
  /proc/self/** r,
}
